#include <metal_stdlib>
using namespace metal;
struct Uniforms {
    float4 display0; // a,c,tx,unused (normalized raw image -> normalized oriented viewport)
    float4 display1; // b,d,ty,unused
    float4 intrinsics;
    float4 imageAndModes; // image width,height,base mode,depth overlay
    float4 rangeAndFlags; // maxDepth,alpha,depth present,confidence present
    float4 colorEncoding; // video range, BT.601 flag,unused,unused
    float4x4 worldToCamera;
};
struct ImageVertex { float4 position [[position]]; float2 uv; };
vertex ImageVertex imageVertex(uint id [[vertex_id]],constant Uniforms &u [[buffer(0)]]) {
    const float2 uv[4] = {float2(0,0),float2(1,0),float2(0,1),float2(1,1)};
    float2 p = uv[id]; float3 q = float3(p,1);
    float2 v = float2(dot(u.display0.xyz,q),dot(u.display1.xyz,q));
    return {float4(v.x*2-1,1-v.y*2,0.999,1),p};
}
float3 heat(float depth,float maximum) {
    float t = clamp(depth/maximum,0.0,1.0);
    // Red near, yellow, cyan, blue far. Same stops as the UI legend.
    if (t < 0.333333) return mix(float3(1,0.12,0.05),float3(1,0.9,0),t*3);
    if (t < 0.666667) return mix(float3(1,0.9,0),float3(0,0.85,1),(t-0.333333)*3);
    return mix(float3(0,0.85,1),float3(0.12,0.15,0.9),(t-0.666667)*3);
}
fragment float4 imageFragment(ImageVertex in [[stage_in]],constant Uniforms &u [[buffer(0)]],
    texture2d<float> luma [[texture(0)]],texture2d<float> chroma [[texture(1)]],
    texture2d<float> depth [[texture(2)]],texture2d<float> confidence [[texture(3)]]) {
    constexpr sampler linearSampler(filter::linear,address::clamp_to_edge);
    constexpr sampler nearestSampler(filter::nearest,address::clamp_to_edge);
    float y = luma.sample(linearSampler,in.uv).r;
    float2 cbcr = chroma.sample(linearSampler,in.uv).rg - float2(128.0/255.0);
    if (u.colorEncoding.x > 0.5) { y = (y-16.0/255.0)*(255.0/219.0); cbcr *= 255.0/224.0; }
    float3 rgb;
    if (u.colorEncoding.y > 0.5) rgb = float3(y+1.402*cbcr.y,y-0.344136*cbcr.x-0.714136*cbcr.y,y+1.772*cbcr.x);
    else rgb = float3(y+1.5748*cbcr.y,y-0.187324*cbcr.x-0.468124*cbcr.y,y+1.8556*cbcr.x);
    rgb = clamp(rgb,0.0,1.0);
    float z = u.rangeAndFlags.z > 0.5 ? depth.sample(nearestSampler,in.uv).r : 0;
    bool valid = isfinite(z) && z > 0;
    float3 invalid = float3(0.12,0.12,0.14);
    float3 depthColor = valid ? heat(z,u.rangeAndFlags.x) : invalid;
    float3 color = rgb;
    if (u.imageAndModes.z > 1.5) {
        if (u.rangeAndFlags.w < 0.5) color = invalid;
        else {
            uint c = uint(round(confidence.sample(nearestSampler,in.uv).r*255));
            color = !valid ? invalid : (c == 2 ? float3(0.1,0.85,0.75) : (c == 1 ? float3(1,0.7,0.1) : float3(0.9,0.2,0.25)));
        }
    } else if (u.imageAndModes.z > 0.5) color = depthColor;
    else if (u.imageAndModes.w > 0.5 && valid) color = mix(rgb,depthColor,u.rangeAndFlags.y);
    return float4(color,1);
}
struct WorldVertex { float4 position; float4 color; };
struct WorldOutput { float4 position [[position]]; float4 color; };
vertex WorldOutput worldVertex(uint id [[vertex_id]],const device WorldVertex *vertices [[buffer(0)]],constant Uniforms &u [[buffer(1)]]) {
    WorldVertex v = vertices[id]; float3 p = (u.worldToCamera * v.position).xyz;
    float z = -p.z;
    // Homogeneous image/display transform lets Metal clip geometry crossing the camera
    // near plane correctly (including the feet connector). Never teleport a behind-camera
    // endpoint to a made-up screen corner: that creates a diagonal streak through the image.
    float3 q = float3((u.intrinsics.x*p.x+(u.intrinsics.z+0.5)*z)/u.imageAndModes.x,
                     (-u.intrinsics.y*p.y+(u.intrinsics.w+0.5)*z)/u.imageAndModes.y,z);
    float2 oriented = float2(dot(u.display0.xyz,q),dot(u.display1.xyz,q));
    float clipZ = (20.0/19.99)*z - (0.2/19.99);
    return {float4(2*oriented.x-z,z-2*oriented.y,clipZ,z),v.color};
}
fragment float4 worldFragment(WorldOutput in [[stage_in]]) { return in.color; }
